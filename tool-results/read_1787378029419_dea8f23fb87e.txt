    46→function formatCurrency(amount: number): string {
    47→  return amount.toLocaleString("en-IN");
    48→}

    50→function validateMobile(value: string): string | undefined {
    51→  const digits = value.replace(/\D/g, "");
    52→  if (digits.length === 0) return "कृपया मोबाइल नंबर दर्ज करें।";
    53→  if (!/^[6-9]/.test(digits)) return "मोबाइल नंबर 6, 7, 8, या 9 से शुरू होना चाहिए।";
    54→  if (digits.length !== 10) return "मोबाइल नंबर 10 अंकों का होना चाहिए।";
    55→  return undefined;
    56→}
    57→
    58→function validateName(value: string): string | undefined {
    59→  const trimmed = value.trim();
    60→  if (trimmed.length === 0) return "कृपया अपना पूरा नाम दर्ज करें।";
    61→  if (trimmed.length < 2) return "नाम कम से कम 2 अक्षर का होना चाहिए।";
    62→  return undefined;
    63→}

   524→
   525→function StepPayment({
   526→  amount,
   527→  donorData,
   528→  onBack,
   529→  onSuccess,
   530→}: {
   531→  amount: number;
   532→  donorData: DonorData;
   533→  onBack: () => void;
   534→  onSuccess: (orderId: string) => void;
   535→}) {
   536→  const [loading, setLoading] = useState(true);
   537→  const [orderError, setOrderError] = useState("");
   538→  const [orderId, setOrderId] = useState<string | null>(null);
   539→  const [upiLink, setUpiLink] = useState("");
   540→  const [qrDataUrl, setQrDataUrl] = useState("");
   541→  const [confirming, setConfirming] = useState(false);
   542→
   543→  // Create order on mount
   544→  useEffect(() => {
   545→    let cancelled = false;
   546→
   547→    async function createOrder() {
   548→      try {
   549→        const res = await fetch("/api/donate/create-order", {
   550→          method: "POST",
   551→          headers: { "Content-Type": "application/json" },
   552→          body: JSON.stringify({
   553→            amount,
   554→            donorName: donorData.donorName.trim(),
   555→            mobile: donorData.mobile,
   556→            email: donorData.email || undefined,
   557→            address: donorData.address || undefined,
   558→            city: donorData.city || undefined,
   559→            district: donorData.district || undefined,
   560→            state: donorData.state || undefined,
   561→            pincode: donorData.pincode || undefined,
   562→          }),
   563→        });
   564→
   565→        const data = await res.json();
   566→
   567→        if (cancelled) return;
   568→
   569→        if (!data.success) {
   570→          setOrderError(data.error || "दान आदेश बनाने में त्रुटि हुई। कृपया पुनः प्रयास करें।");
   571→          setLoading(false);
   572→          return;
   573→        }
   574→
   575→        setOrderId(data.orderId);
   576→        setUpiLink(data.upiLink);
   577→
   578→        // Generate QR code
   579→        const QRCode = await import("qrcode");
   580→        const url = await QRCode.toDataURL(data.upiLink, {
   581→          width: 280,
   582→          margin: 2,
   583→          color: {
   584→            dark: "#5A3A24",
   585→            light: "#FFFFFF",
   586→          },
   587→        });
   588→
   589→        if (!cancelled) {
   590→          setQrDataUrl(url);
   591→          setLoading(false);
   592→        }
   593→      } catch {
   594→        if (!cancelled) {
   595→          setOrderError("भुगतान पृष्ठ लोड करने में त्रुटि हुई। कृपया पुनः प्रयास करें।");
   596→          setLoading(false);
   597→        }
   598→      }
   599→    }
   600→
   601→    createOrder();
   602→    return () => {
   603→      cancelled = true;
   604→    };
   605→  }, [amount, donorData]);
   606→
   607→  const handleConfirmPayment = async () => {
   608→    if (!orderId) return;
   609→    setConfirming(true);
   610→    try {
   611→      await fetch("/api/donate/confirm-payment", {
   612→        method: "POST",
   613→        headers: { "Content-Type": "application/json" },
   614→        body: JSON.stringify({ orderId }),
   615→      });
   616→      onSuccess(orderId);
   617→    } catch {
   618→      setConfirming(false);
   619→    }
   620→  };
   621→
   622→  if (loading) {
   623→    return (
   624→      <div className="flex flex-col items-center justify-center py-16">
   625→        <div
   626→          className="w-10 h-10 border-4 border-light-beige border-t-elegant-orange rounded-full animate-spin"
   627→        />
   628→        <p className="mt-4 text-sm text-muted-brown">भुगतान पृष्ठ तैयार हो रहा है...</p>
   629→      </div>
   630→    );
   631→  }
   632→
   633→  if (orderError) {
   634→    return (
   635→      <div>
   636→        <div className="text-center py-8">
   637→          <p className="text-deep-maroon text-sm font-medium">{orderError}</p>
   638→        </div>
   639→        <button
   640→          type="button"
   641→          onClick={onBack}
   642→          className="
   643→            w-full rounded-full border-2 border-light-gold py-3.5 text-base font-semibold
   644→            text-deep-warm-brown bg-warm-white hover:bg-warm-ivory transition-all duration-200
   645→            active:scale-[0.98]
   646→          "
   647→        >
   648→          वापस जाएं
   649→        </button>
   650→      </div>
   651→    );
   652→  }
   653→
   654→  return (
   655→    <div>
   656→      {/* Summary */}
   657→      <div className="rounded-2xl border-2 border-light-beige bg-warm-white p-4 sm:p-5 mb-6">
   658→        <h2 className="text-base font-semibold text-deep-warm-brown mb-3">
   659→          दान विवरण
   660→        </h2>
   661→        <div className="space-y-2 text-sm">
   662→          <div className="flex justify-between">
   663→            <span className="text-muted-brown">मंदिर</span>
   664→            <span className="text-deep-warm-brown font-medium">
   665→              कृष्णकुंज माँ कर्मा धाम
   666→            </span>
   667→          </div>
   668→          <div className="flex justify-between">
   669→            <span className="text-muted-brown">दानदाता</span>
   670→            <span className="text-deep-warm-brown font-medium">
   671→              {donorData.donorName}
   672→            </span>
   673→          </div>
   674→          <div className="h-px bg-light-beige my-2" />
   675→          <div className="flex justify-between items-center">
   676→            <span className="text-muted-brown">दान राशि</span>
   677→            <span className="text-elegant-orange font-bold text-lg">
   678→              ₹{formatCurrency(amount)}
   679→            </span>
   680→          </div>
   681→        </div>
   682→      </div>
   683→
   684→      {/* QR Code Card */}
   685→      <div className="rounded-2xl border-2 border-light-gold bg-warm-white p-5 sm:p-6 text-center">
   686→        <p className="text-base sm:text-lg font-semibold text-deep-warm-brown mb-4">
   687→          स्कैन करके ₹{formatCurrency(amount)} का दान करें
   688→        </p>
   689→
   690→        {qrDataUrl && (
   691→          <div className="inline-block p-3 rounded-2xl border-2 border-light-gold bg-warm-white shadow-sm">
   692→            <img
   693→              src={qrDataUrl}
   694→              alt={`UPI QR कोड - ₹${formatCurrency(amount)} दान करें`}
   695→              width={240}
   696→              height={240}
   697→              className="w-48 h-48 sm:w-56 sm:h-56 mx-auto"
   698→            />
   699→          </div>
   700→        )}
   701→
   702→        <p className="text-xs text-muted-brown mt-3 leading-relaxed">
   703→          किसी भी UPI ऐप (Google Pay, PhonePe, Paytm आदि) से स्कैन करें
   704→        </p>
   705→      </div>
   706→
   707→      {/* UPI Pay Button */}
   708→      <a
   709→        href={upiLink}
   710→        className="
   711→          mt-5 w-full flex items-center justify-center rounded-full py-3.5 text-base font-semibold
   712→          bg-elegant-orange text-warm-white hover:bg-soft-saffron shadow-md transition-all duration-200
   713→          active:scale-[0.98]
   714→        "
   715→      >
   716→        UPI से भुगतान करें
   717→      </a>
   718→
   719→      {/* Confirm Payment Button */}
   720→      <button
   721→        type="button"
   722→        onClick={handleConfirmPayment}
   723→        disabled={confirming}
   724→        className={`
   725→          mt-3 w-full rounded-full py-3.5 text-base font-semibold transition-all duration-200
   726→          ${
   727→            confirming
   728→              ? "bg-light-beige text-muted-brown/50 cursor-not-allowed"
   729→              : "border-2 border-elegant-orange text-elegant-orange bg-warm-white hover:bg-warm-ivory active:scale-[0.98]"
   730→          }
   731→        `}
   732→      >
   733→        {confirming ? "सत्यापित हो रहा है..." : "मैंने भुगतान कर दिया"}
   734→      </button>
   735→
   736→      {/* Back Button */}
   737→      <button
   738→        type="button"
   739→        onClick={onBack}
   740→        className="
   741→          mt-3 w-full rounded-full border-2 border-light-beige py-3 text-sm font-medium
   742→          text-muted-brown bg-warm-white hover:bg-warm-ivory transition-all duration-200
   743→        "
   744→      >
   745→        वापस जाएं
   746→      </button>
   747→    </div>
   748→  );
   749→}

   751→// ---------------------------------------------------------------------------
   752→// Step 4: Payment Processing
   753→// ---------------------------------------------------------------------------
   754→
   755→function StepProcessing({ orderId }: { orderId: string }) {
   756→  const router = useRouter();
   757→
   758→  useEffect(() => {
   759→    const timer = setTimeout(async () => {
   760→      try {
   761→        await fetch("/api/donate/mark-success", {
   762→          method: "POST",
   763→          headers: { "Content-Type": "application/json" },
   764→          body: JSON.stringify({ orderId }),
   765→        });
   766→      } catch {
   767→        // Even if mark-success fails, still redirect to success page
   768→      }
   769→      router.push(`/donate/success/${orderId}`);
   770→    }, 3000);
   771→
   772→    return () => clearTimeout(timer);
   773→  }, [orderId, router]);
   774→
   775→  return (
   776→    <div className="flex flex-col items-center justify-center py-16">
   777→      <div
   778→        className="w-12 h-12 border-4 border-light-beige border-t-elegant-orange rounded-full animate-spin"
   779→        role="status"
   780→        aria-label="भुगतान सत्यापित हो रहा है"
   781→      />
   782→      <p className="mt-6 text-lg font-semibold text-deep-warm-brown">
   783→        भुगतान सत्यापित किया जा रहा है...
   784→      </p>
   785→      <p className="mt-2 text-sm text-muted-brown">कृपया कुछ क्षण प्रतीक्षा करें।</p>
   786→    </div>
   787→  );
   788→}
