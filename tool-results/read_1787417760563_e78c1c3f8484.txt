    47→    47→function validateMobile(value: string): string | undefined {
    48→    48→  const digits = value.replace(/\D/g, "");
    49→    49→  if (digits.length === 0) return "कृपया मोबाइल नंबर दर्ज करें।";
    50→    50→  if (!/^[6-9]/.test(digits)) return "मोबाइल नंबर 6, 7, 8, या 9 से शुरू होना चाहिए।";
    51→    51→  if (digits.length !== 10) return "मोबाइल नंबर 10 अंकों का होना चाहिए।";
    52→    52→  return undefined;
    53→    53→}

    55→    55→function validateName(value: string): string | undefined {
    56→    56→  const trimmed = value.trim();
    57→    57→  if (trimmed.length === 0) return "कृपया अपना पूरा नाम दर्ज करें।";
    58→    58→  if (trimmed.length < 2) return "नाम कम से कम 2 अक्षर का होना चाहिए।";
    59→    59→  return undefined;
    60→    60→}

   565→   565→        });

   567→   567→          setQrDataUrl(qrUrl);
   568→   568→          setLoading(false);
   569→   569→        }

   573→   573→          setLoading(false);
   574→   574→        }
   575→   575→      }
   576→   576→    }
   577→   577→
   578→   578→    init();
   579→   579→    return () => { cancelled = true; };
   580→   580→  // eslint-disable-next-line react-hooks/exhaustive-deps
   581→   581→  }, []);

   583→   583→  const handleConfirmPayment = async () => {

   587→   587→      await fetch("/api/donate/confirm-payment", {
   588→   588→        method: "POST",
   589→   589→        headers: { "Content-Type": "application/json" },
   590→   590→        body: JSON.stringify({ orderId }),
   591→   591→      });

   593→   593→    } catch {

   595→   595→    }
   596→   596→  };

   598→   598→  if (loading) {

   600→   600→      <div className="flex flex-col items-center justify-center py-16">
   601→   601→        <div
   602→   602→          className="w-10 h-10 border-4 border-light-beige border-t-elegant-orange rounded-full animate-spin"
   603→   603→        />
   604→   604→        <p className="mt-4 text-sm text-muted-brown">भुगतान पृष्ठ तैयार हो रहा है...</p>
   605→   605→      </div>
   606→   606→    );
   607→   607→  }
   608→   608→
   609→   609→  if (orderError) {
   610→   610→    return (
   611→   611→      <div>
   612→   612→        <div className="text-center py-8">
   613→   613→          <p className="text-deep-maroon text-sm font-medium">{orderError}</p>
   614→   614→        </div>
   615→   615→        <button
   616→   616→          type="button"
   617→   617→          onClick={onBack}
   618→   618→          className="
   619→   619→            w-full rounded-full border-2 border-light-gold py-3.5 text-base font-semibold
   620→   620→            text-deep-warm-brown bg-warm-white hover:bg-warm-ivory transition-all duration-200
   621→   621→            active:scale-[0.98]
   622→   622→          "
   623→   623→        >
   624→   624→          वापस जाएं
   625→   625→        </button>
   626→   626→      </div>
   627→   627→    );
   628→   628→  }
   629→   629→
   630→   630→  return (
   631→   631→    <div>
   632→   632→      {/* Summary */}
   633→   633→      <div className="rounded-2xl border-2 border-light-beige bg-warm-white p-4 sm:p-5 mb-6">
   634→   634→        <h2 className="text-base font-semibold text-deep-warm-brown mb-3">
   635→   635→          दान विवरण
   636→   636→        </h2>
   637→   637→        <div className="space-y-2 text-sm">
   638→   638→          <div className="flex justify-between">
   639→   639→            <span className="text-muted-brown">मंदिर</span>
   640→   640→            <span className="text-deep-warm-brown font-medium">
   641→   641→              कृष्णकुंज माँ कर्मा धाम
   642→   642→            </span>
   643→   643→          </div>
   644→   644→          <div className="flex justify-between">
   645→   645→            <span className="text-muted-brown">दानदाता</span>

   647→   647→              {donorData.donorName}
   648→   648→            </span>
   649→   649→          </div>
   650→   650→          <div className="h-px bg-light-beige my-2" />
   651→   651→          <div className="flex justify-between items-center">
   652→   652→            <span className="text-muted-brown">दान राशि</span>

   654→   654→              ₹{formatCurrency(amount)}

   663→   663→          UPI QR कोड स्कैन करें

   735→   735→function StepProcessing({ orderId }: { orderId: string }) {
   736→   736→  const router = useRouter();

   738→   738→  useEffect(() => {
   739→   739→    const timer = setTimeout(async () => {

   742→   742→          method: "POST",
   743→   743→          headers: { "Content-Type": "application/json" },
   744→   744→          body: JSON.stringify({ orderId }),
   745→   745→        });
   746→   746→      } catch {
   747→   747→        // Even if mark-success fails, still redirect to success page
   748→   748→      }
   749→   749→      router.push(`/donate/success/${orderId}`);
   750→   750→    }, 3000);
   751→   751→
   752→   752→    return () => clearTimeout(timer);
   753→   753→  }, [orderId, router]);
   754→   754→
   755→   755→  return (
   756→   756→    <div className="flex flex-col items-center justify-center py-16">
   757→   757→      <div

   768→   768→}
