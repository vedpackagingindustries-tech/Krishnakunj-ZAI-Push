    36→function formatCurrency(amount: number): string {
    37→  return amount.toLocaleString("en-IN");
    38→}
    39→
    40→function validateMobile(value: string): string | undefined {
    41→  const digits = value.replace(/\D/g, "");
    42→  if (digits.length === 0) return "कृपया मोबाइल नंबर दर्ज करें।";
    43→  if (!/^[6-9]/.test(digits)) return "मोबाइल नंबर 6, 7, 8, या 9 से शुरू होना चाहिए।";
    44→  if (digits.length !== 10) return "मोबाइल नंबर 10 अंकों का होना चाहिए।";
    45→  return undefined;
    46→}
    47→
    48→function validateName(value: string): string | undefined {
    49→  const trimmed = value.trim();
    50→  if (trimmed.length === 0) return "कृपया अपना पूरा नाम दर्ज करें।";
    51→  if (trimmed.length < 2) return "नाम कम से कम 2 अक्षर का होना चाहिए।";
    52→  return undefined;
    53→}

   261→function StepDetails({
   262→  amount,
   263→  donorData,
   264→  setDonorData,
   265→  errors,
   266→  setErrors,
   267→  onBack,
   268→  onProceed,
   269→}: {
   270→  amount: number;
   271→  donorData: DonorData;
   272→  setDonorData: React.Dispatch<React.SetStateAction<DonorData>>;
   273→  errors: FieldErrors;
   274→  setErrors: React.Dispatch<React.SetStateAction<FieldErrors>>;
   275→  onBack: () => void;
   276→  onProceed: () => void;
   277→}) {
   278→  const [consent, setConsent] = useState(false);
   279→
   280→  const updateField = (field: keyof DonorData, value: string) => {
   281→    setDonorData((prev) => ({ ...prev, [field]: value }));
   282→    // Clear field error on change
   283→    if (field === "donorName" || field === "mobile" || field === "address") {
   284→      setErrors((prev) => ({ ...prev, [field]: undefined }));
   285→    }
   286→  };
   287→
   288→  const handleProceed = () => {
   289→    const newErrors: FieldErrors = {};
   290→    const nameErr = validateName(donorData.donorName);
   291→    if (nameErr) newErrors.donorName = nameErr;
   292→
   293→    const mobileErr = validateMobile(donorData.mobile);
   294→    if (mobileErr) newErrors.mobile = mobileErr;
   295→
   296→    if (!donorData.address.trim()) {
   297→      newErrors.address = "कृपया अपना पता दर्ज करें।";
   298→    }
   299→
   300→    if (!consent) {
   301→      newErrors.consent = "कृपया ऊपर दिए गए कथन पर सहमति दें।";
   302→    }
   303→
   304→    if (Object.keys(newErrors).length > 0) {
   305→      setErrors(newErrors);
   306→      return;
   307→    }
   308→
   309→    onProceed();
   310→  };
   311→
   312→  const inputClass = `
   313→    w-full rounded-xl border-2 border-light-beige bg-warm-white text-deep-warm-brown
   314→    placeholder:text-muted-brown/50
   315→    focus:border-soft-saffron focus:outline-none focus:ring-2 focus:ring-soft-saffron/30
   316→    px-4 py-3 text-base transition-colors
   317→  `;
   318→
   319→  const labelClass = "block text-sm font-medium text-muted-brown mb-1.5";
   320→  const errorClass = "text-sm text-deep-maroon mt-1";
   321→  const requiredMark = <span className="text-deep-maroon">*</span>;
   322→
   323→  return (
   324→    <div>
   325→      {/* Amount Display */}
   326→      <div className="text-center mb-6">
   327→        <p className="text-sm text-muted-brown mb-1">आपका दान</p>
   328→        <p className="text-3xl sm:text-4xl font-bold text-elegant-orange">
   329→          ₹{formatCurrency(amount)}
   330→        </p>
   331→      </div>
   332→
   333→      <OrnamentalDivider />
   334→
   335→      {/* Required Fields */}
   336→      <div className="space-y-4 mt-4">
   337→        <div>
   338→          <label htmlFor="donorName" className={labelClass}>
   339→            पूरा नाम{requiredMark}
   340→          </label>
   341→          <input
   342→            id="donorName"
   343→            type="text"
   344→            placeholder="अपना पूरा नाम लिखें"
   345→            value={donorData.donorName}
   346→            onChange={(e) => updateField("donorName", e.target.value)}
   347→            className={`${inputClass} ${errors.donorName ? "border-deep-maroon" : ""}`}
   348→          />
   349→          {errors.donorName && <p className={errorClass}>{errors.donorName}</p>}
   350→        </div>
   351→
   352→        <div>
   353→          <label htmlFor="mobile" className={labelClass}>
   354→            मोबाइल नंबर{requiredMark}
   355→          </label>
   356→          <input
   357→            id="mobile"
   358→            type="tel"
   359→            maxLength={10}
   360→            placeholder="10 अंकों का मोबाइल नंबर"
   361→            value={donorData.mobile}
   362→            onChange={(e) => {
   363→              const digits = e.target.value.replace(/\D/g, "").slice(0, 10);
   364→              updateField("mobile", digits);
   365→            }}
   366→            className={`${inputClass} ${errors.mobile ? "border-deep-maroon" : ""}`}
   367→          />
   368→          {errors.mobile && <p className={errorClass}>{errors.mobile}</p>}
   369→        </div>
   370→
   371→        {/* Address — Required */}
   372→        <div>
   373→          <label htmlFor="address" className={labelClass}>
   374→            पता{requiredMark}
   375→          </label>
   376→          <input
   377→            id="address"
   378→            type="text"
   379→            placeholder="मकान नंबर, गली / मोहल्ला"
   380→            value={donorData.address}
   381→            onChange={(e) => updateField("address", e.target.value)}
   382→            className={`${inputClass} ${errors.address ? "border-deep-maroon" : ""}`}
   383→          />
   384→          {errors.address && <p className={errorClass}>{errors.address}</p>}
   385→        </div>
   386→
   387→
   388→      </div>
   389→
   390→      {/* Consent Checkbox */}
   391→      <div className="mt-6 flex items-start gap-3">
   392→        <Checkbox
   393→          id="consent"
   394→          checked={consent}
   395→          onCheckedChange={(checked) => {
   396→            setConsent(checked === true);
   397→            if (checked === true) {
   398→              setErrors((prev) => ({ ...prev, consent: undefined }));
   399→            }
   400→          }}
   401→          className="mt-0.5 data-[state=checked]:bg-elegant-orange data-[state=checked]:border-elegant-orange"
   402→        />
   403→        <label
   404→          htmlFor="consent"
   405→          className="text-sm text-deep-warm-brown leading-relaxed cursor-pointer select-none"
   406→        >
   407→          मैं मंदिर निर्माण हेतु अपनी श्रद्धानुसार दान कर रहा/रही हूँ।
   408→        </label>
   409→      </div>
   410→      {errors.consent && (
   411→        <p className={errorClass}>{errors.consent}</p>
   412→      )}
   413→
   414→      {/* Buttons */}
   415→      <div className="flex gap-3 mt-8">
   416→        <button
   417→          type="button"
   418→          onClick={onBack}
   419→          className="
   420→            flex-1 rounded-full border-2 border-light-gold py-3.5 text-base font-semibold
   421→            text-deep-warm-brown bg-warm-white hover:bg-warm-ivory transition-all duration-200
   422→            active:scale-[0.98]
   423→          "
   424→        >
   425→          वापस जाएं
   426→        </button>
   427→        <button
   428→          type="button"
   429→          onClick={handleProceed}
   430→          className="
   431→            flex-[2] rounded-full py-3.5 text-base font-semibold transition-all duration-200
   432→            bg-elegant-orange text-warm-white hover:bg-soft-saffron shadow-md active:scale-[0.98]
   433→          "
   434→        >
   435→          भुगतान के लिए आगे बढ़ें
   436→        </button>
   437→      </div>
   438→    </div>
   439→  );
   440→}

   446→function StepPayment({
   447→  amount,
   448→  donorData,
   449→  existingOrderId,
   450→  onBack,
   451→  onSuccess,
   452→}: {
   453→  amount: number;
   454→  donorData: DonorData;
   455→  existingOrderId: string;
   456→  onBack: () => void;
   457→  onSuccess: (orderId: string) => void;
   458→}) {
   459→  const [loading, setLoading] = useState(!existingOrderId);
   460→  const [orderError, setOrderError] = useState("");
   461→  const [orderId, setOrderId] = useState<string | null>(existingOrderId || null);
   462→  const [upiLink, setUpiLink] = useState("");
   463→  const [qrDataUrl, setQrDataUrl] = useState("");
   464→  const [confirming, setConfirming] = useState(false);
   465→
   466→  // Create order & QR on mount (skip if already created)
   467→  useEffect(() => {
   468→    let cancelled = false;
   469→
   470→    async function init() {
   471→      try {
   472→        let link = '';
   473→
   474→        if (existingOrderId) {
   475→          const params = new URLSearchParams({
   476→            pa: 'sahubhagwat392@indianbk',
   477→            pn: 'कृष्णकुंज माँ कर्मा धाम',
   478→            am: String(amount),
   479→            cu: 'INR',
   480→            tn: `मंदिर निर्माण दान - ₹${amount}`,
   481→          });
   482→          link = `upi://pay?${params.toString()}`;
   483→          setUpiLink(link);
   484→        } else {
   485→          const res = await fetch('/api/donate/create-order', {
   486→            method: 'POST',
   487→            headers: { 'Content-Type': 'application/json' },
   488→            body: JSON.stringify({
   489→              amount,
   490→              donorName: donorData.donorName.trim(),
   491→              mobile: donorData.mobile,
   492→              address: donorData.address.trim() || undefined,
   493→            }),
   494→          });
   495→          const data = await res.json();
   496→          if (cancelled) return;
   497→          if (!data.success) {
   498→            setOrderError(data.error || 'दान आदेश बनाने में त्रुटि हुई। कृपया पुनः प्रयास करें।');
   499→            setLoading(false);
   500→            return;
   501→          }
   502→          setOrderId(data.orderId);
   503→          link = data.upiLink;
   504→          setUpiLink(link);
   505→        }
   506→
   507→        // Generate QR code with exact amount
   508→        const QRCode = await import('qrcode');
   509→        const qrUrl = await QRCode.toDataURL(link, {
   510→          width: 280,
   511→          margin: 2,
   512→          color: { dark: '#5A3A24', light: '#FFFFFF' },
   513→        });
   514→        if (!cancelled) {
   515→          setQrDataUrl(qrUrl);
   516→          setLoading(false);
   517→        }
   518→      } catch {
   519→        if (!cancelled) {
   520→          setOrderError('भुगतान पृष्ठ लोड करने में त्रुटि हुई। कृपया पुनः प्रयास करें।');
   521→          setLoading(false);
   522→        }
   523→      }
   524→    }
   525→
   526→    init();
   527→    return () => { cancelled = true; };
   528→  // eslint-disable-next-line react-hooks/exhaustive-deps
   529→  }, []);
   530→
   531→  const handleConfirmPayment = async () => {
   532→    if (!orderId) return;
   533→    setConfirming(true);
   534→    try {
   535→      await fetch("/api/donate/confirm-payment", {
   536→        method: "POST",
   537→        headers: { "Content-Type": "application/json" },
   538→        body: JSON.stringify({ orderId }),
   539→      });
   540→      onSuccess(orderId);
   541→    } catch {
   542→      setConfirming(false);
   543→    }
   544→  };
   545→
   546→  if (loading) {
   547→    return (
   548→      <div className="flex flex-col items-center justify-center py-16">
   549→        <div
   550→          className="w-10 h-10 border-4 border-light-beige border-t-elegant-orange rounded-full animate-spin"
   551→        />
   552→        <p className="mt-4 text-sm text-muted-brown">भुगतान पृष्ठ तैयार हो रहा है...</p>
   553→      </div>
   554→    );
   555→  }
   556→
   557→  if (orderError) {
   558→    return (
   559→      <div>
   560→        <div className="text-center py-8">
   561→          <p className="text-deep-maroon text-sm font-medium">{orderError}</p>
   562→        </div>
   563→        <button
   564→          type="button"
   565→          onClick={onBack}
   566→          className="
   567→            w-full rounded-full border-2 border-light-gold py-3.5 text-base font-semibold
   568→            text-deep-warm-brown bg-warm-white hover:bg-warm-ivory transition-all duration-200
   569→            active:scale-[0.98]
