    40→    40→function validateMobile(value: string): string | undefined {
    41→    41→  const digits = value.replace(/\D/g, "");
    42→    42→  if (digits.length === 0) return "कृपया मोबाइल नंबर दर्ज करें।";
    43→    43→  if (!/^[6-9]/.test(digits)) return "मोबाइल नंबर 6, 7, 8, या 9 से शुरू होना चाहिए।";
    44→    44→  if (digits.length !== 10) return "मोबाइल नंबर 10 अंकों का होना चाहिए।";
    45→    45→  return undefined;
    46→    46→}

    48→    48→function validateName(value: string): string | undefined {
    49→    49→  const trimmed = value.trim();
    50→    50→  if (trimmed.length === 0) return "कृपया अपना पूरा नाम दर्ज करें।";
    51→    51→  if (trimmed.length < 2) return "नाम कम से कम 2 अक्षर का होना चाहिए।";
    52→    52→  return undefined;
    53→    53→}
