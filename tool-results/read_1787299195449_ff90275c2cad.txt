   169→/* ─── Temple bell hook: loops user's MP3 continuously ─── */
   170→function useTempleBell() {
   171→  const [bellPlaying, setBellPlaying] = useState(false);
   172→  const audioRef = useState<HTMLAudioElement | null>(null);
   173→  const audio = audioRef[0];
   174→  const setAudio = audioRef[1];
   175→  const fadeRef = useState<NodeJS.Timeout | null>(null);
   176→  const fadeTimer = fadeRef[0];
   177→  const setFadeTimer = fadeRef[1];
   178→
   179→  const fadeIn = useCallback(() => {
   180→    if (!audio) return;
   181→    if (fadeTimer) clearInterval(fadeTimer);
   182→    const t = setInterval(() => {
   183→      if (!audio) { clearInterval(t); return; }
   184→      if (audio.volume < 0.6) {
   185→        audio.volume = Math.min(audio.volume + 0.03, 0.6);
   186→      } else {
   187→        clearInterval(t);
   188→      }
   189→    }, 100);
   190→    setFadeTimer(t);
   191→  }, [audio, fadeTimer, setFadeTimer]);
   192→
   193→  const toggleBell = useCallback(async () => {
   194→    try {
   195→      if (!audio) {
   196→        const a = new Audio("/audio/temple-bell.mp3");
   197→        a.loop = true;
   198→        a.volume = 0;
   199→        setAudio(a);
   200→        await a.play();
   201→        setBellPlaying(true);
   202→        // start fade after audio is set
   203→        setTimeout(() => {
   204→          const t = setInterval(() => {
   205→            if (a.volume < 0.6) {
   206→              a.volume = Math.min(a.volume + 0.03, 0.6);
   207→            } else {
   208→              clearInterval(t);
   209→            }
   210→          }, 100);
   211→          setFadeTimer(t);
   212→        }, 50);
   213→      } else if (audio.paused) {
   214→        audio.volume = 0;
   215→        await audio.play();
   216→        setBellPlaying(true);
   217→        fadeIn();
   218→      } else {
   219→        // fade out then pause
   220→        if (fadeTimer) clearInterval(fadeTimer);
   221→        const t = setInterval(() => {
   222→          if (audio.volume > 0.05) {
   223→            audio.volume = Math.max(audio.volume - 0.05, 0);
   224→          } else {
   225→            audio.pause();
   226→            clearInterval(t);
   227→            setBellPlaying(false);
   228→          }
   229→        }, 50);
   230→      }
   231→    } catch {
   232→      // audio blocked
   233→    }
   234→  }, [audio, fadeTimer, setAudio, setFadeTimer, fadeIn]);
   235→
   236→  // Cleanup on unmount
   237→  useEffect(() => {
   238→    return () => {
   239→      if (audio) { audio.pause(); audio.src = ""; }
   240→      if (fadeTimer) clearInterval(fadeTimer);
   241→    };
   242→  }, []);
   243→
   244→  return { bellPlaying, toggleBell };
   245→}
