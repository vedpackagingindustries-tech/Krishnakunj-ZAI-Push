    65→// ── Helpers ─────────────────────────────────────────
    66→
    67→function formatCurrency(n: number): string {
    68→  return '₹' + n.toLocaleString('en-IN')
    69→}
    70→
    71→function formatDate(iso: string): string {
    72→  return new Date(iso).toLocaleDateString('hi-IN', {
    73→    day: '2-digit',
    74→    month: 'short',
    75→    year: 'numeric',
    76→  })
    77→}
    78→
    79→function getStatusBadge(status: string) {
    80→  switch (status) {
    81→    case 'SUCCESS':
    82→      return (
    83→        <span className="inline-flex items-center rounded-full bg-green-100 px-2.5 py-0.5 text-xs font-medium text-green-800">
    84→          सफल
    85→        </span>
    86→      )
    87→    case 'PENDING':
    88→      return (
    89→        <span className="inline-flex items-center rounded-full bg-yellow-100 px-2.5 py-0.5 text-xs font-medium text-yellow-800">
    90→          PENDING
    91→        </span>
    92→      )
    93→    case 'FAILED':
    94→      return (
    95→        <span className="inline-flex items-center rounded-full bg-red-100 px-2.5 py-0.5 text-xs font-medium text-red-800">
    96→          FAILED
    97→        </span>
    98→      )
    99→    case 'CANCELLED':
   100→      return (
   101→        <span className="inline-flex items-center rounded-full bg-orange-100 px-2.5 py-0.5 text-xs font-medium text-orange-800">
   102→          CANCELLED
   103→        </span>
   104→      )
   105→    default:
   106→      return (
   107→        <span className="inline-flex items-center rounded-full bg-light-beige px-2.5 py-0.5 text-xs font-medium text-muted-brown">
   108→          {status}
   109→        </span>
   110→      )
   111→  }
   112→}

   150→// ── Detail Dialog ───────────────────────────────────
   151→
   152→function DonationDetailDialog({
   153→  donation,
   154→  open,
   155→  onOpenChange,
   156→}: {
   157→  donation: Donation | null
   158→  open: boolean
   159→  onOpenChange: (open: boolean) => void
   160→}) {
   161→  if (!donation) return null
   162→
   163→  const details = [
   164→    { label: 'पावती क्रमांक', value: donation.receiptNumber },
   165→    { label: 'दानदाता', value: donation.donorName },
   166→    { label: 'मोबाइल', value: donation.mobile },
   167→    { label: 'ईमेल', value: donation.email || '—' },
   168→    { label: 'पता', value: [donation.address, donation.city, donation.district, donation.state, donation.pincode].filter(Boolean).join(', ') || '—' },
   169→    { label: 'राशि', value: formatCurrency(donation.amount) },
   170→    { label: 'भुगतान माध्यम', value: donation.paymentMethod || '—' },
   171→    { label: 'Transaction ID', value: donation.transactionId || '—' },
   172→    { label: 'स्थिति', value: donation.paymentStatus, badge: true },
   173→    { label: 'निर्माण तिथि', value: formatDate(donation.createdAt) },
   174→    { label: 'भुगतान तिथि', value: donation.paidAt ? formatDate(donation.paidAt) : '—' },
   175→  ]
   176→
   177→  return (
   178→    <Dialog open={open} onOpenChange={onOpenChange}>
   179→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   180→        <DialogHeader>
   181→          <DialogTitle className="text-deep-warm-brown">
   182→            दान विवरण
   183→          </DialogTitle>
   184→          <DialogDescription className="text-muted-brown">
   185→            पावती क्रमांक: {donation.receiptNumber}
   186→          </DialogDescription>
   187→        </DialogHeader>
   188→        <div className="space-y-3 mt-2">
   189→          {details.map((item) => (
   190→            <div
   191→              key={item.label}
   192→              className="flex flex-col sm:flex-row sm:items-start gap-1 sm:gap-3 border-b border-light-beige/50 pb-2 last:border-0 last:pb-0"
   193→            >
   194→              <span className="text-xs font-medium text-muted-brown shrink-0 sm:w-32">
   195→                {item.label}
   196→              </span>
   197→              <span className="text-sm text-deep-warm-brown font-medium">
   198→                {item.badge ? getStatusBadge(item.value) : item.value}
   199→              </span>
   200→            </div>
   201→          ))}
   202→        </div>
   203→      </DialogContent>
   204→    </Dialog>
   205→  )
   206→}

   229→
   230→  const limit = 20
   231→
   232→  // ── Fetch donations ──
   233→  const fetchDonations = useCallback(async () => {

   270→

   274→    setPage(1)
   275→  }
   276→
   277→  // ── Status change ──
   278→  const handleStatusChange = (val: string) => {
   279→    setStatus(val === 'all' ? '' : val)
   280→    setPage(1)
   281→  }
   282→
   283→  // ── Export (blob download with auth header) ──
   284→  const handleExport = async (format: 'csv' | 'excel') => {
   285→    const params = new URLSearchParams()
   286→    params.set('format', format)
   287→    if (search) params.set('search', search)
   288→    if (status) params.set('status', status)
   289→    if (dateFrom) params.set('dateFrom', dateFrom)
   290→    if (dateTo) params.set('dateTo', dateTo)
   291→    if (amountMin) params.set('amountMin', amountMin)
   292→    if (amountMax) params.set('amountMax', amountMax)

   297→      const token = getAdminToken()

   300→        headers: { Authorization: `Bearer ${token}` },

   309→      document.body.appendChild(a)
   310→      a.click()

   314→      // handled silently

   356→            size="sm"

   358→            onClick={() => handleExport('excel')}
   359→          >

   361→            Excel
   362→          </Button>
   363→        </div>
   364→      </div>

   366→      {/* Search Bar */}
