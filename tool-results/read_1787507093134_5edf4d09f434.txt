    43→function formatCurrency(amount: number): string {
    44→  return amount.toLocaleString("en-IN");
    45→}
    46→
    47→function validateMobile(value: string): string | undefined {
    48→  const digits = value.replace(/\D/g, "");
    49→  if (digits.length === 0) return "कृपया मोबाइल नंबर दर्ज करें।";
    50→  if (!/^[6-9]/.test(digits)) return "मोबाइल नंबर 6, 7, 8, या 9 से शुरू होना चाहिए।";
    51→  if (digits.length !== 10) return "मोबाइल नंबर 10 अंकों का होना चाहिए।";
    52→  return undefined;
    53→}
    54→
    55→function validateName(value: string): string | undefined {
    56→  const trimmed = value.trim();
    57→  if (trimmed.length === 0) return "कृपया अपना पूरा नाम दर्ज करें।";
    58→  if (trimmed.length < 2) return "नाम कम से कम 2 अक्षर का होना चाहिए।";
    59→  return undefined;
    60→}

   268→function StepDetails({
   269→  amount,
   270→  donorData,
   271→  setDonorData,
   272→  errors,
   273→  setErrors,
   274→  onBack,
   275→  onProceed,
   276→}: {
   277→  amount: number;
   278→  donorData: DonorData;
   279→  setDonorData: React.Dispatch<React.SetStateAction<DonorData>>;
   280→  errors: FieldErrors;
   281→  setErrors: React.Dispatch<React.SetStateAction<FieldErrors>>;
   282→  onBack: () => void;
   283→  onProceed: () => void;
   284→}) {
   285→  const [consent, setConsent] = useState(false);
   286→
   287→  const updateField = (field: keyof DonorData, value: string) => {
   288→    setDonorData((prev) => ({ ...prev, [field]: value }));
   289→    // Clear field error on change
   290→    if (field === "donorName" || field === "mobile" || field === "address") {
   291→      setErrors((prev) => ({ ...prev, [field]: undefined }));
   292→    }
   293→  };
   294→
   295→  const handleProceed = () => {
   296→    const newErrors: FieldErrors = {};
   297→    const nameErr = validateName(donorData.donorName);
   298→    if (nameErr) newErrors.donorName = nameErr;
   299→
   300→    const mobileErr = validateMobile(donorData.mobile);
   301→    if (mobileErr) newErrors.mobile = mobileErr;
   302→
   303→    if (!donorData.address.trim()) {
   304→      newErrors.address = "कृपया अपना पता दर्ज करें।";
   305→    }
   306→
   307→    if (!consent) {
   308→      newErrors.consent = "कृपया ऊपर दिए गए कथन पर सहमति दें।";
   309→    }
   310→
   311→    if (Object.keys(newErrors).length > 0) {
   312→      setErrors(newErrors);
   313→      return;
   314→    }
   315→
   316→    onProceed();
   317→  };
   318→
   319→  const inputClass = `
   320→    w-full rounded-xl border-2 border-light-beige bg-warm-white text-deep-warm-brown
   321→    placeholder:text-muted-brown/50
   322→    focus:border-soft-saffron focus:outline-none focus:ring-2 focus:ring-soft-saffron/30

   333→      <div className="text-center mb-6">
   334→        <p className="text-sm text-muted-brown mb-1">आपका दान</p>
   335→        <p className="text-3xl sm:text-4xl font-bold text-elegant-orange">
   336→          ₹{formatCurrency(amount)}
   337→        </p>
   338→      </div>
   339→
   340→      <OrnamentalDivider />
   341→
   342→      {/* Required Fields */}
   343→      <div className="space-y-4 mt-4">
   344→        <div>
   345→          <label htmlFor="donorName" className={labelClass}>
   346→            पूरा नाम{requiredMark}
   347→          </label>
   348→          <input
   349→            id="donorName"
   350→            type="text"
   351→            placeholder="अपना पूरा नाम लिखें"
   352→            value={donorData.donorName}
   353→            onChange={(e) => updateField("donorName", e.target.value)}
   354→            className={`${inputClass} ${errors.donorName ? "border-deep-maroon" : ""}`}
   355→          />
   356→          {errors.donorName && <p className={errorClass}>{errors.donorName}</p>}
   357→        </div>
   358→
   359→        <div>
   360→          <label htmlFor="mobile" className={labelClass}>
   361→            मोबाइल नंबर{requiredMark}
   362→          </label>
   363→          <input
   364→            id="mobile"
   365→            type="tel"
   366→            maxLength={10}

   368→            value={donorData.mobile}
   369→            onChange={(e) => {
   370→              const digits = e.target.value.replace(/\D/g, "").slice(0, 10);
   371→              updateField("mobile", digits);
   372→            }}
   373→            className={`${inputClass} ${errors.mobile ? "border-deep-maroon" : ""}`}
   374→          />
   375→          {errors.mobile && <p className={errorClass}>{errors.mobile}</p>}
   376→        </div>
   377→
   378→        {/* Address — Required */}
   379→        <div>
   380→          <label htmlFor="address" className={labelClass}>
   381→            पता{requiredMark}
   382→          </label>
   383→          <input
   384→            id="address"
   385→            type="text"
   386→            placeholder="मकान नंबर, गली / मोहल्ला"
   387→            value={donorData.address}
   388→            onChange={(e) => updateField("address", e.target.value)}
   389→            className={`${inputClass} ${errors.address ? "border-deep-maroon" : ""}`}
   390→          />
   391→          {errors.address && <p className={errorClass}>{errors.address}</p>}
   392→        </div>
   393→
   394→        {/* Optional Fields */}
   395→        <div className="grid grid-cols-2 gap-3">
   396→          <div>
   397→            <label htmlFor="city" className={labelClass}>
   398→              शहर
   399→            </label>
   400→            <input
   401→              id="city"
   402→              type="text"
   403→              placeholder="शहर"
   404→              value={donorData.city}
   405→              onChange={(e) => updateField("city", e.target.value)}
   406→              className={inputClass}
   407→            />
   408→          </div>
   409→          <div>
   410→            <label htmlFor="district" className={labelClass}>

   414→              id="district"
   415→              type="text"
   416→              placeholder="जिला"

   495→function StepPayment({
   496→  amount,
   497→  donorData,
   498→  existingOrderId,
   499→  onBack,
   500→  onSuccess,
   501→}: {
   502→  amount: number;
   503→  donorData: DonorData;
   504→  existingOrderId: string;
   505→  onBack: () => void;
   506→  onSuccess: (orderId: string) => void;
   507→}) {
   508→  const [loading, setLoading] = useState(!existingOrderId);
   509→  const [orderError, setOrderError] = useState("");
   510→  const [orderId, setOrderId] = useState<string | null>(existingOrderId || null);
   511→  const [upiLink, setUpiLink] = useState("");
   512→  const [qrDataUrl, setQrDataUrl] = useState("");
   513→  const [confirming, setConfirming] = useState(false);
   514→
   515→  // Create order & QR on mount (skip if already created)
   516→  useEffect(() => {
   517→    let cancelled = false;
   518→
   519→    async function init() {
   520→      try {
   521→        let link = '';
   522→
   523→        if (existingOrderId) {
   524→          const params = new URLSearchParams({
   525→            pa: 'sahubhagwat392@indianbk',
   526→            pn: 'कृष्णकुंज माँ कर्मा धाम',
   527→            am: String(amount),
   528→            cu: 'INR',
   529→            tn: `मंदिर निर्माण दान - ₹${amount}`,
   530→          });
   531→          link = `upi://pay?${params.toString()}`;
   532→          setUpiLink(link);
   533→        } else {
   534→          const res = await fetch('/api/donate/create-order', {
   535→            method: 'POST',
   536→            headers: { 'Content-Type': 'application/json' },
   537→            body: JSON.stringify({
   538→              amount,
   539→              donorName: donorData.donorName.trim(),
   540→              mobile: donorData.mobile,
   541→              address: donorData.address.trim() || undefined,
   542→              city: donorData.city || undefined,
   543→              district: donorData.district || undefined,
   544→              state: donorData.state || undefined,
   545→            }),
   546→          });
   547→          const data = await res.json();
   548→          if (cancelled) return;
   549→          if (!data.success) {
   550→            setOrderError(data.error || 'दान आदेश बनाने में त्रुटि हुई। कृपया पुनः प्रयास करें।');
   551→            setLoading(false);
   552→            return;
   553→          }
   554→          setOrderId(data.orderId);
   555→          link = data.upiLink;
   556→          setUpiLink(link);
   557→        }
   558→
   559→        // Generate QR code with exact amount
   560→        const QRCode = await import('qrcode');
   561→        const qrUrl = await QRCode.toDataURL(link, {
   562→          width: 280,
   563→          margin: 2,
   564→          color: { dark: '#5A3A24', light: '#FFFFFF' },
   565→        });
   566→        if (!cancelled) {
   567→          setQrDataUrl(qrUrl);
   568→          setLoading(false);
   569→        }
   570→      } catch {
   571→        if (!cancelled) {
   572→          setOrderError('भुगतान पृष्ठ लोड करने में त्रुटि हुई। कृपया पुनः प्रयास करें।');
   573→          setLoading(false);
   574→        }
   575→      }
   576→    }
   577→
   578→    init();

   734→
   735→function StepProcessing({ orderId }: { orderId: string }) {
   736→  const router = useRouter();
   737→
   738→  useEffect(() => {
   739→    const timer = setTimeout(async () => {
   740→      try {
   741→        await fetch("/api/donate/mark-success", {
   742→          method: "POST",
   743→          headers: { "Content-Type": "application/json" },
   744→          body: JSON.stringify({ orderId }),
   745→        });
   746→      } catch {
   747→        // Even if mark-success fails, still redirect to success page
   748→      }
   749→      router.push(`/donate/success/${orderId}`);
   750→    }, 3000);
   751→
   752→    return () => clearTimeout(timer);
   753→  }, [orderId, router]);
   754→
   755→  return (
   756→    <div className="flex flex-col items-center justify-center py-16">
   757→      <div
   758→        className="w-12 h-12 border-4 border-light-beige border-t-elegant-orange rounded-full animate-spin"
   759→        role="status"
   760→        aria-label="भुगतान सत्यापित हो रहा है"
   761→      />
   762→      <p className="mt-6 text-lg font-semibold text-deep-warm-brown">
   763→        भुगतान सत्यापित किया जा रहा है...
   764→      </p>
   765→      <p className="mt-2 text-sm text-muted-brown">कृपया कुछ क्षण प्रतीक्षा करें।</p>
   766→    </div>
   767→  );
   768→}
